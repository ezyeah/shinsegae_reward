  1. fold list 기본 구조

    .fold-wrap > fold-line-list / fold-box-list > fold-item > fold-header | fold-content

    .fold-wrap > fold-overflow-box > fold-content | fold-header

    *주로 단일 형태로 사용

  1. 주로 단일 컨텐츠 노출/미노출로 사용

                        <div class="fold-wrap">
                            <!-- fold-overflow-box -->
                            <div class="fold-overflow-box">
                                <!-- fold-content -->
                                <div class="fold-content">
                                    <div style="background-color: rgba(255, 0, 0, .2); height: 200px;">컨텐츠 영역 확인을 위한 임시 div입니다.</div>
                                </div>
                                <!-- // fold-content -->
                                <!-- fold-header -->
                                <div class="fold-header">
                                    <button type="button" class="btn-fold">
                                        <span class="text">자세히 보기</span>
                                        <span class="ico-gray-arrow18"></span>
                                    </button>
                                </div>
                                <!-- // fold-header -->
                            </div>
                            <!-- // fold-overflow-box -->
                        </div>
